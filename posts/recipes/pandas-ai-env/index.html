로컬 LLM으로 데이터 해석하기#

import pandas as pd
import requests

df = pd.read_csv("data.csv")
summary = df.describe().to_string()

# Ollama 로컬 LLM에 전달
response = requests.post(
    "http://localhost:11434/api/generate",
    json={
        "model": "qwen3:4b-instruct",
        "prompt": f"다음 데이터 요약을 분석해줘:\n{summary}",
        "stream": False
    }
)
print(response.json()["response"])
