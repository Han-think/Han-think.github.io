1. LangGraph Loop: writer와 critic 연결 이해하기#

무엇을 배웠나#

LangGraph의 Loop는 단순한 while 반복문이 아니라, State를 유지한 채 Node 간 Edge가 연결된 구조라는 점을 명확히 이해하게 되었습니다. 특히 router가 결정하는 경로에 따라 이전 Node인 writer로 돌아가는 방식이, 일반적인 코드 흐름과 어떻게 다른지 감을 잡는 데 도움이 되었습니다. writer, critic, router의 역할 분담이 명확해지면서 전체 파이프라인 구조가 훨씬 직관적이고 관리하기 쉬워졌다는 느낌을 받았습니다.

헷갈렸던 점#

처음에는 LangGraph의 Loop가 일반적인 while 반복문과 같은 것인지, 아니면 단순히 Node 사이의 연결 고리인지 혼란스러웠습니다. 특히 router가 결정한 경로에 따라 이전 Node인 writer로 다시 돌아가는 방식이 기존 프로그래밍의 흐름과는 다른 구조라서 감 잡는 데 시간이 걸렸습니다. 결국 LangGraph에서의 Loop는 while문처럼 코드를 반복하는 것이 아니라, State를 유지한 채 Edge가 이전 Node를 다시 가리키는 연결 구조임을 이해하게 되었습니다.

정리된 이해#

LangGraph에서의 Loop는 while문과 같은 단순한 반복문이 아니라, State를 유지한 채 Edge가 이전 Node를 다시 가리키는 구조입니다. 핵심은 데이터가 담긴 State가 각 노드를 거칠 때마다 업데이트되며 전달되고, 라우터의 결정에 따라 특정 조건이 충족될 때까지 writer와 critic이 순환한다는 점입니다. 이를 통해 콘텐츠 생성과 검토라는 역할을 명확히 분리하고, 상태 관리의 복잡성을 Edge 연결로 효율적으로 해결할 수 있습니다.

from typing import Literal
from langgraph.graph import StateGraph, START, END

# 상태 정의 (State의 지속성 확보)
class WriteState(dict):
    topic: str
    draft: str = ""
    feedback: str = ""
    state: str = "drafting"  # drafting 또는 reviewing
    count: int = 0           # 반복 횟수 추적

# 노드 정의
def writer(state: WriteState) -> dict:
    """콘텐츠를 생성합니다."""
    print(f"[Writer] '{state['topic]}'에 대한 초안을 작성합니다... (count: {state['count']})")
    # 간단한 초안 생성 로직 (실제로는 LLM 호출)
    new_draft = f"초안: {state['topic']}에 대해 논의해 봅시다."
    return {"draft": new_draft, "count": state.get("count", 0) + 1}

def critic(state: WriteState) -> dict:
    """초안을 검토하고 피드백을 제공합니다."""
    print(f"[Critic] 초안을 검토합니다...")
    # 간단한 피드백 로직 (실제로는 LLM 기반 평가)
    if "오류" in state['topic'] or state['count'] > 2:
        feedback = "검토 완료: 개선 필요 없음 또는 최대 반복 횟수 도달."
        return {"feedback": feedback, "state": "reviewing"}
    else:
        feedback = "피드백: 더 자세한 내용이나 수정이 필요할 수 있습니다."
        return {"feedback": feedback, "state": "drafting"}

def router(state: WriteState) -> Literal["END", "writer"]:
    """다음 단계(END 또는 writer)로 라우팅합니다."""
    print(f"[Router] 현재 상태: {state['state']}, 반복 횟수: {state['count']}")

    # 종료 조건: 긍정적인 피드백이거나 최대 반복 횟수 초과 시 END
    if state["state"] == "reviewing" or state["count"] >= 3:
        print("[Router] 경로: END")
        return END
    else:
        print("[Router] 경로: writer (Loop)")
        # Loop를 위해 'writer' 노드로 라우팅
        return "writer"

# 그래프 빌드
workflow = StateGraph(WriteState)

# 노드 추가
workflow.add_node("writer", writer)
workflow.add_node("critic", critic)

# 엣지 추가 (Loop 구조 형성)
# START -> writer
workflow.add_edge(START, "writer")
# writer -> critic
workflow.add_edge("writer", "critic")
# critic -> router (conditonal edge 로직 시뮬레이션을 위한 단순 연결)
# 여기서는 router 함수의 반환값에 따라 분기됨
workflow.add_conditional_edges(
    "critic",
    router,
    {
        "writer": "writer",  # Loop: critic의 결과가 writer를 가리키면 반복
        END: END             # 종료 시 END로 이동
    }
)

app = workflow.compile()

# 실행 시뮬레이션 (초기 상태)
initial_state = WriteState(topic="인공지능의 윤리적 이슈", count=0)

print("--- LangGraph Loop 실행 시작 ---")
for output in app.stream(initial_state):
    print(output)
print("--- 종료 ---")

2. Critic JSON 판정과 결정론적 가드레일: LLM 평가와 Python 검증 결합#

무엇을 배웠나#

LLM의 판단은 창의적이지만 형식 오류를 범할 수 있어, Python 기반의 deterministic guardrail을 결합해 핵심 키워드와 구조를 검증해야 한다. 특히 JSON 파싱 실패 시 무조건 재시도하고, 최대 3회 이내에 해결하지 못하면 종료하는 등 retry 횟수 제한으로 안정성을 확보한다.

헷갈렸던 점#

처음에는 LLM이 스스로 판단해서 끝내면 되지 않을까 싶었는데, 모델이 형식이나 기준을 지키지 못할 위험이 있다는 점이 와닿았습니다. 결국 LLM의 평가와 Python 기반의 검증을 결합해야 하며, 중요한 문구나 구조는 코드가 안전장치처럼 체크하는 것이 가장 안정적이라고 이해했습니다.

정리된 이해#

LLM의 판단은 창의적이지만 형식 오류가 발생할 수 있으므로, Python 기반의 결정론적 검증을 결합해 핵심 키워드와 구조를 반드시 확인해야 한다. 이를 통해 LLM의 평가 결과와 코드의 안전장치를 연계하여 최종 판단을 내리고, 재시도 횟수 제한으로 무한 루프를 방지하는 것이 안정적인 시스템 설계의 핵심이다.

import json
from typing import TypedDict, Literal

# 1. 상태 정의 (State)
class State(TypedDict):
    draft: str  # LLM이 생성한 초안
    feedback: str  # 비평가(Llm)의 피드백
    iteration: int  # 재시도 횟수
    status: Literal["working", "retry", "finish"]  # 현재 상태

# 2. 결정론적 가드레일 (Deterministic Guardrail)
def check_keywords(state: State) -> bool:
    """
    LLM의 피드백이나 초안에 필수 키워드가 포함되어 있는지 Python 코드로 확인.
    이 함수가 False를 반환하면 무조건 retry가 발생합니다.
    """
    # 예시: "혁신" 또는 "미래"라는 단어가 반드시 들어있어야 함
    draft_text = state.get("draft", "")
    feedback_text = state.get("feedback", "")

    # 핵심 검증 로직: 키워드 존재 여부 확인
    has_innovation = "혁신" in draft_text or "미래" in draft_text
    return has_innovation

# 3. 메인 루프 로직 (Critic JSON 판정 및 결정론적 가드레일 적용)
def run_critic_loop(initial_draft: str, max_iterations: int = 3) -> State:
    state = {
        "draft": initial_draft,
        "feedback": "",
        "iteration": 0,
        "status": "working"
    }

    while state["status"] == "working" and state["iteration"] < max_iterations:
        state["iteration"] += 1

        # --- LLM 평가 단계 (비판적 사고) ---
        # 여기서는 가상의 LLM 호출 및 JSON 파싱 결과를 가정합니다.
        # 실제로는 model.invoke(state["draft"]) 등을 호출합니다.
        print(f"--- Iteration {state['iteration']}: LLM evaluating draft ---")

        # 시뮬레이션: LLM이 JSON 형식 불량을 일으키거나, 키워드가 없을 수 있음
        # 예시 피드백 (실제로는 LLM 출력 파싱)
        simulated_feedback = "내용은 좋으나 '혁신'이라는 키워드가 부족합니다."
        state["feedback"] = simulated_feedback

        # JSON 파싱 실패 시 안전장치 (핵심 로직)
        try:
            # 모델이 반환한 텍스트가 JSON 형태라고 가정하고 파싱 시도
            # 만약 model output이 유효하지 않은 JSON라면 이 부분에서 예외 발생
            parsed_json = json.loads(state["draft"])
        except json.JSONDecodeError:
            print("⚠️ JSON 파싱 실패! 형식을 다시 검사합니다.")
            state["status"] = "retry"
            continue # 루프의 다음 단계로 넘어가지 않고 retry로 전환

        # --- 결정론적 가드레일 적용 ---
        # LLM의 결과가 even if LLM says PASS, we check the rules.
        if not check_keywords(state):
            print("⚠️ 키워드 검증 실패 (예: '혁신' 누락). 재시도합니다.")
            state["status"] = "retry"
            continue

        # 모든 조건을 만족하면 종료
        print("✅ 모든 가드레일을 통과했습니다. 작업을 완료합니다.")
        state["status"] = "finish"
        break

    if state["iteration"] >= max_iterations:
        state["status"] = "finish"
        print("❌ 최대 재시도 횟수 초과로 종료.")

    return state

# 실행 예시
if __name__ == "__main__":
    # 키워드가 없는 초안으로 시작 (의도적으로 실패 시키기 위함)
    bad_draft = "이번 프로젝트는 기존 방식을 유지합니다."

    final_state = run_critic_loop(bad_draft)
    print(f"\n최종 상태: {final_state['status']} (반복 횟수: {final_state['iteration']})")

3. LangGraph 툴 호출의 3단계 파이프라인: 판단-실행-전달#

무엇을 배웠나#

LLM은 직접 함수를 실행하는 것이 아니라, 도구가 필요하다고 판단해 요청을 보낸 뒤 별도의 Python 코드가 계산을 수행한 결과를 다시 받아야 합니다. 이 과정 자체가 LLM과 Tool 사이에서 반복되는 순환 구조를 형성하기 때문에, LangGraph에서의 Tool Calling은 하나의 그래프 흐름으로 이해할 수 있습니다.

헷갈렸던 점#

LLM이 직접 함수를 실행하는 것이 아니라, Tool Node라는 별도 파이프라인에서 Python 코드를 수행한다는 점이 가장 큰 혼란이었습니다. 또한 LLM과 도구 사이의 연결고리가 단순히 ‘함수 호출’이 아니라, 결과가 메시지 형태로 다시 전달되는 구조임을 간과하고 있었습니다.

정리된 이해#

LangGraph에서 Tool Calling은 LLM이 직접 함수를 실행하는 것이 아니라, LLM이 도구 사용을 판단하고 별도의 Python 코드가 실제 계산을 수행한 뒤 결과를 전달하는 3단계 파이프라인 구조입니다. Agent가 필요성을 판단하면 Tool Node가 함수를 실행하고, 그 결과는 ToolMessage를 통해 다시 Agent로 돌아와 최종 응답을 생성합니다.

# LangGraph Tool Calling 흐름 예시 (개념도)

from typing import Annotated, Sequence
from langgraph.graph import StateGraph, add_messages
from langgraph.prebuilt import ToolNode, tools_condition
from langchain_core.messages import HumanMessage, AIMessage, ToolMessage
from langchain_core.tools import tool

# 1. 도구 정의 (Tool)
@tool
def multiply(x: int, y: int) -> int:
    """두 수를 곱합니다."""
    return x * y

# 2. 상태 그래프 정의
# messages 상태는 대화 기록을 관리합니다.
State = Annotated[Sequence[BaseMessage], add_messages]

graph = StateGraph(State)

# 에이전트 노드: LLM이 도구 사용 여부를 판단
def agent(state: State):
    last_message = state[-1]
    # LLM이 multiply 툴을 호출하도록 결정한 경우
    if "multiply" in last_message.content.lower():
        return {"messages": [ToolMessage(tool="multiply", content=f"{last_message.content} 계산 요청")]}
    else:
        return {"messages": [AIMessage(content="계산기가 필요하지 않습니다.")]}

# 그래프에 노드 추가
graph.add_node("agent", agent)
# 툴 실행 노드: 실제 Python 함수 실행
graph.add_node("multiply_tool", ToolNode([multiply]))

# 엣지(연결선) 정의
# 사용자 입력 -> 에이전트
graph.set_entry_point("agent")
# 에이전트 -> 툴 (조건부 엣지, 도구 필요 시)
graph.add_conditional_edges(
    "agent",
    tools_condition,  # LLM의 출력에 따라 도구 실행 여부 결정
    {"tools": "multiply_tool", END": END}
)
# 툴 결과 -> 에이전트 (결과가 돌아오면 다시 에이전트가 최종 답변 생성)
graph.add_edge("multiply_tool", "agent")

app = graph.compile()

# 3. 실행 시뮬레이션
user_input = "5 곱하기 3은?"
print(f"User: {user_input}")

# 그래프 실행 (시뮬레이션을 위해 단순화된 입력 처리)
# 실제 LangGraph는 Runnable 방식으로 호출됩니다.
messages = [HumanMessage(content=user_input)]
output_state = app.invoke({"messages": messages})

print("\n--- 최종 결과 ---")
print(output_state["messages"][-1].content)

총평#

오늘 오전 전체에서 배운 것#

LangGraph Loop는 State를 유지한 채 Node 간 Edge가 연결된 구조이며, router에 의해 결정되는 경로에 따라 이전 Node로 복귀하는 방식이 일반적인 코드 흐름과 다르다. LLM의 판단은 창의적일 수 있어 Python 기반의 deterministic guardrail을 결합해 핵심 키워드와 구조를 검증하고, JSON 파싱 실패 시 최대 3회 이내 재시도 후 종료하여 안정성을 확보한다. 또한 LLM이 직접 함수를 실행하는 것이 아니라 도구 사용을 판단해 요청을 보낸 뒤 별도의 Python 코드가 결과를 반환하는 3단계 파이프라인 구조를 통해 LangGraph에서의 Tool Calling을 하나의 그래프 흐름으로 이해할 수 있었다.

다음에 조심할 것#

LLM의 판단은 창의적이지만 형식 오류를 범할 수 있어, Python 기반의 deterministic guardrail을 결합해 핵심 키워드와 구조를 검증해야 한다. 특히 JSON 파싱 실패 시 무조건 재시도하고, 최대 3회 이내에 해결하지 못하면 종료하는 등 retry 횟수 제한으로 안정성을 확보한다. LangGraph에서의 Tool Calling은 LLM이 도구를 호출하고 결과를 받아오는 판단-실행-전달의 순환 구조를 하나의 그래프 흐름으로 이해해야 하며, 단순한 while 반복문이 아니라 State를 유지한 채 Node 간 Edge가 연결된 구조임을 명심해야 한다.

다음 복습 주제#

LangGraph Loop 구조에서 State 기반 라우팅이 어떻게 반복 제어를 담당하는지, 그리고 LLM의 불확실성을 Python 검증으로 보완하는 안정적인 파이프라인 설계 원리를 중심으로 심화 학습할 계획입니다. 특히 Tool Calling의 판단-실행-전달 흐름이 그래프 내에서의 상태 전파와 어떻게 맞물리는지 구체적인 시나리오를 통해 재정리하겠습니다.
