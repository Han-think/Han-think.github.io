SQL 실습 20문제#

S01. 제품 라인별 제품 수#

products에서 productLine별 제품 수를 구하고 많은 순서로 정렬하시오.

SELECT productLine, COUNT(*) AS product_count FROM products GROUP BY productLine ORDER BY product_count DESC;

S02. 국가별 고객 수#

customers에서 country별 고객 수를 구하시오.

SELECT country, COUNT(*) AS customer_count FROM customers GROUP BY country ORDER BY customer_count DESC;

S03. 2004년 주문 건수#

orders에서 2004년 주문 건수를 구하시오.

SELECT COUNT(*) AS order_count_2004 FROM orders WHERE strftime('%Y', orderDate) = '2004';

S04. 제품 라인별 평균 MSRP#

productLine별 평균 MSRP를 소수 둘째 자리까지 구하시오.

SELECT productLine, ROUND(AVG(MSRP), 2) AS avg_msrp FROM products GROUP BY productLine ORDER BY avg_msrp DESC;

S05. 국가별 총 주문 금액#

customers, orders, orderdetails를 연결해 국가별 총 주문 금액을 구하시오.

SELECT c.country, ROUND(SUM(od.quantityOrdered * od.priceEach), 2) AS total_sales FROM customers AS c JOIN orders AS o ON c.customerNumber = o.customerNumber JOIN orderdetails AS od ON o.orderNumber = od.orderNumber GROUP BY c.country ORDER BY total_sales DESC;

S06. 제품 라인별 매출과 수량#

products와 orderdetails를 연결해 productLine별 판매수량과 매출을 구하시오.

SELECT p.productLine, SUM(od.quantityOrdered) AS total_quantity, ROUND(SUM(od.quantityOrdered * od.priceEach), 2) AS total_sales FROM products AS p JOIN orderdetails AS od ON p.productCode = od.productCode GROUP BY p.productLine ORDER BY total_sales DESC;

S07. HAVING으로 3회 이상 구매 고객#

Classic Cars를 3회 이상 구매한 고객을 찾으시오.

SELECT c.customerNumber, c.customerName, COUNT(DISTINCT o.orderNumber) AS order_count FROM customers AS c JOIN orders AS o ON c.customerNumber = o.customerNumber JOIN orderdetails AS od ON o.orderNumber = od.orderNumber JOIN products AS p ON od.productCode = p.productCode WHERE p.productLine = 'Classic Cars' GROUP BY c.customerNumber, c.customerName HAVING COUNT(DISTINCT o.orderNumber) >= 3 ORDER BY order_count DESC;

S08. 평균보다 신용한도가 높은 고객#

전체 평균 creditLimit보다 높은 고객을 조회하시오.

SELECT customerNumber, customerName, creditLimit FROM customers WHERE creditLimit > (SELECT AVG(creditLimit) FROM customers) ORDER BY creditLimit DESC;

S09. 2003년과 2004년 모두 주문한 고객#

2003년에도 주문하고 2004년에도 주문한 고객을 찾으시오.

SELECT c.customerNumber, c.customerName FROM customers AS c WHERE c.customerNumber IN (SELECT customerNumber FROM orders WHERE strftime('%Y', orderDate) = '2003') AND c.customerNumber IN (SELECT customerNumber FROM orders WHERE strftime('%Y', orderDate) = '2004');

S10. 한 번도 주문하지 않은 고객#

orders에 없는 고객을 LEFT JOIN으로 찾으시오.

SELECT c.customerNumber, c.customerName FROM customers AS c LEFT JOIN orders AS o ON c.customerNumber = o.customerNumber WHERE o.orderNumber IS NULL;

S11. 고객별 결제액 순위 TOP 10#

payments에서 고객별 총액과 RANK를 구하고 TOP 10을 출력하시오.

SELECT customerNumber, total_amount, RANK() OVER (ORDER BY total_amount DESC) AS pay_rank FROM (SELECT customerNumber, SUM(amount) AS total_amount FROM payments GROUP BY customerNumber) AS t ORDER BY pay_rank LIMIT 10;

S12. 국가 내 신용한도 순위#

국가별 creditLimit 순위를 구하시오.

SELECT country, customerName, creditLimit, RANK() OVER (PARTITION BY country ORDER BY creditLimit DESC) AS country_credit_rank FROM customers ORDER BY country, country_credit_rank;

S13. 월별 매출과 전월 대비 증감#

월별 매출과 LAG 기반 전월 대비 증감을 구하시오.

WITH monthly_sales AS (SELECT strftime('%Y-%m', o.orderDate) AS order_month, SUM(od.quantityOrdered * od.priceEach) AS sales FROM orders AS o JOIN orderdetails AS od ON o.orderNumber = od.orderNumber GROUP BY strftime('%Y-%m', o.orderDate)) SELECT order_month, ROUND(sales, 2) AS sales, ROUND(LAG(sales) OVER (ORDER BY order_month), 2) AS prev_sales, ROUND(sales - LAG(sales) OVER (ORDER BY order_month), 2) AS diff_sales FROM monthly_sales ORDER BY order_month;

S14. 판매되지 않은 제품#

orderdetails에 등장하지 않은 제품을 NOT EXISTS로 찾으시오.

SELECT p.productCode, p.productName FROM products AS p WHERE NOT EXISTS (SELECT 1 FROM orderdetails AS od WHERE od.productCode = p.productCode);

S15. 주문 상태별 건수#

orders에서 status별 주문 건수를 구하시오.

SELECT status, COUNT(*) AS order_count FROM orders GROUP BY status ORDER BY order_count DESC;

S16. CASE로 신용한도 등급#

creditLimit이 100000 이상 high, 70000 이상 mid, 나머지 low로 표시하시오.

SELECT customerName, creditLimit, CASE WHEN creditLimit >= 100000 THEN 'high' WHEN creditLimit >= 70000 THEN 'mid' ELSE 'low' END AS credit_group FROM customers ORDER BY creditLimit DESC;

S17. NTILE로 4구간 나누기#

creditLimit 기준으로 고객을 4개 구간으로 나누시오.

SELECT customerName, creditLimit, NTILE(4) OVER (ORDER BY creditLimit DESC) AS credit_quartile FROM customers ORDER BY credit_quartile, creditLimit DESC;

S18. 제품별 평균 주문 단가#

orderdetails에서 productCode별 평균 priceEach를 구하시오.

SELECT productCode, ROUND(AVG(priceEach), 2) AS avg_price FROM orderdetails GROUP BY productCode ORDER BY avg_price DESC;

S19. CTE로 국가별 매출 TOP 3#

CTE로 국가별 매출을 만든 뒤 상위 3개 국가를 출력하시오.

WITH country_sales AS (SELECT c.country, SUM(od.quantityOrdered * od.priceEach) AS total_sales FROM customers AS c JOIN orders AS o ON c.customerNumber = o.customerNumber JOIN orderdetails AS od ON o.orderNumber = od.orderNumber GROUP BY c.country) SELECT country, ROUND(total_sales, 2) AS total_sales FROM country_sales ORDER BY total_sales DESC LIMIT 3;

S20. UNION ALL로 연도별 주문 고객 합치기#

2003년과 2004년 고객번호를 year_label과 함께 합치시오.

SELECT '2003' AS year_label, customerNumber FROM orders WHERE strftime('%Y', orderDate) = '2003' UNION ALL SELECT '2004' AS year_label, customerNumber FROM orders WHERE strftime('%Y', orderDate) = '2004';
